## 第 1 题

```python
import torch

def make_fk_samples(count, seed, q1=(-1., 1.), q2=(-1.2, 1.2)):
    if not isinstance(count, int) or count < 1:
        raise ValueError("positive sample count required")
    bounds = torch.tensor([q1, q2], dtype=torch.float32)
    if bounds.shape != (2, 2) or not torch.isfinite(bounds).all():
        raise ValueError("two finite ranges required")
    if not (bounds[:, 1] > bounds[:, 0]).all():
        raise ValueError("increasing ranges required")
    rng = torch.Generator().manual_seed(seed)
    q = bounds[:, 0] + torch.rand(count, 2, generator=rng) * (bounds[:, 1]-bounds[:, 0])
    x = .3*torch.cos(q[:, 0]) + .2*torch.cos(q.sum(dim=1))
    y = .3*torch.sin(q[:, 0]) + .2*torch.sin(q.sum(dim=1))
    return q, torch.stack([x, y], dim=1)

def fit_standardizer(x_train):
    if x_train.ndim != 2 or len(x_train) < 2 or not torch.isfinite(x_train).all():
        raise ValueError("finite training matrix with at least two samples required")
    mean = x_train.mean(dim=0).detach().clone()
    scale = x_train.std(dim=0, unbiased=False).clamp_min(1e-6).detach().clone()
    return mean, scale

def standardize(x, mean, scale):
    if x.ndim != 2 or mean.shape != (x.shape[1],) or scale.shape != mean.shape:
        raise ValueError("feature shapes must match")
    if not torch.isfinite(scale).all() or not (scale > 0).all():
        raise ValueError("positive finite scales required")
    return (x - mean) / scale
```

这里输入是 CPU float32。真实项目还要检查全部数据的 dtype/device、有限值和字段语义。验证集单独标准化会改变同一数值对应的模型输入，并使预处理依赖验证分布。

## 第 2 题

```python
def run_epoch(model, loader, loss_fn, optimizer=None, device="cpu"):
    training = optimizer is not None
    model.train(training)
    total_loss, count = 0., 0
    with torch.set_grad_enabled(training):
        for x, y in loader:
            x, y = x.to(device), y.to(device)
            if training:
                optimizer.zero_grad(set_to_none=True)
            pred = model(x)
            loss = loss_fn(pred, y)
            if loss.ndim != 0 or not torch.isfinite(loss):
                raise ValueError("finite scalar mean loss required")
            if training:
                loss.backward()
                optimizer.step()
            total_loss += loss.detach().item() * len(x)
            count += len(x)
    if count == 0:
        raise ValueError("nonempty loader required")
    return total_loss / count
```

前提：loss_fn 返回每样本同等权重的均值，且每个样本元素数相同。MSE 可接 ML-01 的 checked_mse；分类用未加权、无 ignore 的 CrossEntropy。变长 token、类别权重、忽略标签需要按真实有效元素统计，不能机械乘 batch。

验证不传 optimizer；用独立、不洗牌的 loader。训练返回值是“各批更新时”的平均损失，画固定参数 train/val 曲线再分别 eval 一遍。

使用 CUDA 时先 `model.to("cuda")`，再构造 optimizer，调用 run_epoch 时显式传 `device="cuda"`。loader 仍可产出 CPU Tensor；搬运发生在循环内。这个教学循环每批调用 item/有限性检查，方便排错，但会同步 GPU；测高吞吐时不能把它当作优化过的训练框架。

## 第 3 题

```python
import copy

def snapshot_cpu(model, optimizer, next_epoch, config, mean, scale, loader_rng):
    if any(p.device.type != "cpu" for p in model.parameters()):
        raise ValueError("this reference covers CPU only")
    return {
        "model": copy.deepcopy(model.state_dict()),
        "optimizer": copy.deepcopy(optimizer.state_dict()),
        "next_epoch": int(next_epoch),
        "config": copy.deepcopy(config),
        "mean": mean.detach().cpu().clone(),
        "scale": scale.detach().cpu().clone(),
        "torch_rng": torch.get_rng_state().clone(),
        "loader_rng": loader_rng.get_state().clone(),
    }

def restore_cpu(payload, model, optimizer, loader_rng):
    model.load_state_dict(payload["model"])
    optimizer.load_state_dict(payload["optimizer"])
    loader_rng.set_state(payload["loader_rng"])
    # 在重建模型/优化器等所有随机初始化之后，最后恢复全局 RNG。
    torch.set_rng_state(payload["torch_rng"])
    return payload["next_epoch"], payload["mean"], payload["scale"]
```

config 只放可核对的基本类型；调用 `torch.save(payload,path)`，读取 `torch.load(path,map_location="cpu",weights_only=True)`。推理用模型、mean/scale 与结构配置；无需恢复 optimizer/RNG，但必须 eval。模型构造由 config 指定，你自己实现；本函数不会替你检查架构、数据文件或预处理是否匹配。

## 第 4 题

构造相同数据；为 DataLoader 显式传入 `generator=torch.Generator().manual_seed(...)`，num_workers=0。保存只在一整轮 iterator 消耗完之后；重建相同模型、Adam、loader，再调用 restore_cpu。继续前不额外取训练 batch 或做随机前向。

在相同 CPU/依赖和这个边界下，可核对连续与恢复的 loss/权重一致。只载入模型能恢复预测，但 Adam 的动量、随机 Dropout 和洗牌顺序会使后续更新走不同路径。Python/NumPy 随机增强、CUDA RNG、多 worker、scheduler、AMP scaler 等若参与训练，还须额外保存相应状态。

### 单 GPU / AMP 的状态片段

以下扩展仍只负责 epoch 边界状态，不替代训练工程。支持 CPU 或一块 CUDA 设备；本章没有随机增强、scheduler 与多 worker。和 CPU 入门片段一样，保存时模型快照有独立存储。

```python
def training_device(model):
    devices = {p.device for p in model.parameters()} | {b.device for b in model.buffers()}
    if len(devices) != 1:
        raise ValueError("one model device required")
    device = next(iter(devices))
    if device.type not in ("cpu", "cuda"):
        raise ValueError("CPU or single CUDA device required")
    return device

def snapshot_training(model, optimizer, next_epoch, config, mean, scale, loader_rng, scaler=None):
    device = training_device(model)
    if scaler is not None and device.type != "cuda":
        raise ValueError("this AMP reference covers CUDA only")
    return {
        "model": copy.deepcopy(model.state_dict()),
        "optimizer": copy.deepcopy(optimizer.state_dict()),
        "next_epoch": int(next_epoch),
        "config": copy.deepcopy(config),
        "mean": mean.detach().cpu().clone(),
        "scale": scale.detach().cpu().clone(),
        "device_type": device.type,
        "device_index": device.index,
        "torch_rng": torch.get_rng_state().clone(),
        "loader_rng": loader_rng.get_state().clone(),
        "cuda_rng": torch.cuda.get_rng_state(device).cpu().clone() if device.type == "cuda" else None,
        "amp_scaler": copy.deepcopy(scaler.state_dict()) if scaler is not None else None,
    }

def restore_training(payload, model, optimizer, loader_rng, scaler=None):
    device = training_device(model)
    if device.type != payload["device_type"] or device.index != payload["device_index"]:
        raise ValueError("same backend/device required for this continuation check")
    if (scaler is None) != (payload["amp_scaler"] is None):
        raise ValueError("matching AMP configuration required")
    model.load_state_dict(payload["model"])
    optimizer.load_state_dict(payload["optimizer"])
    if scaler is not None:
        scaler.load_state_dict(payload["amp_scaler"])
    loader_rng.set_state(payload["loader_rng"].cpu())
    # 模型和优化器等对象全部重建后，最后恢复随机状态。
    torch.set_rng_state(payload["torch_rng"].cpu())
    if device.type == "cuda":
        torch.cuda.set_rng_state(payload["cuda_rng"].cpu(), device)
    return payload["next_epoch"], payload["mean"], payload["scale"]
```

仍用 `torch.load(path,map_location="cpu",weights_only=True)` 读自建文件，再重建相同结构、移动模型、创建 optimizer/scaler，最后 restore_training。PyTorch 2.6 的 optimizer.load_state_dict 会处理与参数关联状态的设备转换；本次 Adam 在 GPU 新进程续训中实际核对，不能把所有状态张量不加区分地 .cuda()。

返回的 mean/scale 在 CPU：在 CPU 预处理再搬 batch，或显式把统计量迁移后在 GPU 标准化。CPU/GPU 间仅恢复预测可以另行搬权重，但不属于本片段承诺的“同一训练路径”。

继续选模时还须保存 best_val/best 模型和日志；可把历史标量放 config，历史最佳权重另存，不用“最后模型”冒充“最佳模型”。本片段的跨进程核对见[记录](../示例核对记录.md#gpu-resume)。
